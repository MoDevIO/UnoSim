 * This includes pin modes, digital pin states, and math constants
 */

// Arduino specific types
typedef bool boolean;
#define byte uint8_t

#define LOW  0x0
#define INPUT 0x0

#define INPUT_PULLUP 0x2
#define LED_BUILTIN 13
 
// Analog pins
#define A0 14
#define A1 15
#define A2 16

#define A4 18
